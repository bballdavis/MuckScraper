{#- Page navigation shared by All/Grouped Stories, Headlines, Op-Ed, Recently Read and Search. Usage: {% from 'pagination.html' import pager %} ... {{ pager(page, total_pages) }} One rounded bar: Previous, the page numbers, Next. The numbers are the first and last page and the ones either side of the current page; a gap is "…", which opens a small box to type a page number (Enter to go). Phones get Previous, "6 of 120" and Next. Links and the jump box keep every other query parameter of the current page (filters, sort, search), changing only `page`. Routes clamp an out-of-range page to the nearest real one. Styles live in theme.css. -#} {% macro pager(page, total_pages) -%} {%- if total_pages > 1 -%} {%- set args = request.args.to_dict() -%} {%- set _ = args.pop('page', None) -%} {%- set near = range([page - 1, 2]|max, [page + 1, total_pages - 1]|min + 1)|list -%} {%- set ns = namespace(pages=[1]) -%} {%- if near and near[0] > 3 %}{% set ns.pages = ns.pages + [None] %}{% elif near and near[0] == 3 %}{% set ns.pages = ns.pages + [2] %}{% endif -%} {%- set ns.pages = ns.pages + near -%} {%- if near and near[-1] < total_pages - 2 %}{% set ns.pages = ns.pages + [None] %}{% elif near and near[-1] == total_pages - 2 %}{% set ns.pages = ns.pages + [total_pages - 1] %}{% endif -%} {%- set ns.pages = ns.pages + [total_pages] -%} {%- set chevron_left = '' -%} {%- set chevron_right = '' -%}
{%- if page > 1 %} {{ chevron_left|safe }}Previous {%- else %} {{ chevron_left|safe }}Previous {%- endif %}
    {%- for p in ns.pages %} {%- if p is none %}
  1. {%- for key, value in args.items() %} {%- endfor %}
    2. {%- elif p == page %}
  2. {{ '{:,}'.format(p) }}
    3. {%- else %}
  3. {{ '{:,}'.format(p) }}
    4. {%- endif %} {%- endfor %}
{{ '{:,}'.format(page) }} of {{ '{:,}'.format(total_pages) }} {%- if page < total_pages %} Next{{ chevron_right|safe }} {%- else %} Next{{ chevron_right|safe }} {%- endif %}
{%- endif -%} {%- endmacro %}
